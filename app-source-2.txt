  }).join('');

function renderCalendar() {
  const events = new Map();
  for (const bill of state.bills.filter((item) => item.dueDate?.startsWith('2026-08'))) {
    const day = Number(bill.dueDate.slice(-2));
    const list = events.get(day) || [];
    list.push(bill);
    events.set(day, list);
  }
  const cells = [];
  for (let blank = 0; blank < 6; blank++) cells.push('<div class="calendar-day empty"></div>');
  for (let day = 1; day <= 31; day++) {
    const dayEvents = events.get(day) || [];
    cells.push(`<div class="calendar-day"><div class="calendar-date">${day}</div>${dayEvents.map((bill) => `<div class="calendar-event ${classifyBill(bill, state.asOfDate)}">${escapeHtml(bill.name)}<br>${formatMoney(bill.amountCents)}</div>`).join('')}</div>`);
  }
  $('#calendar-grid').innerHTML = cells.join('');
  const sorted = [...state.bills].sort((a,b) => (a.dueDate || '9999').localeCompare(b.dueDate || '9999'));
  $('#agenda-list').innerHTML = `<div class="section-heading"><h3>August agenda</h3><span class="badge">${state.bills.length} items</span></div>` + sorted.map((bill) => `<div class="agenda-item"><span>${categoryIcon(bill.category)}</span><div><strong>${escapeHtml(bill.name)}</strong><div>${bill.dueDate ? formatDate(bill.dueDate) : escapeHtml(bill.dueLabel || 'Recurring')}</div></div><strong>${formatMoney(bill.amountCents)}</strong></div>`).join('');
}

      <small>${box.targetCents > 0 ? `${progress}% of ${formatMoney(box.targetCents)}` : 'Set a goal'}</small>

}

function queuePlan() {
  const allocations = allocateAvailableFunds(state.bills, state.availableCents, state.asOfDate).allocations;
  const next = buildBankStops(allocations, state.savings);
  if (currentStops.length && activeStopIndex > 0) pendingStops = next;
  else { currentStops = next; activeStopIndex = 0; }
}

function startLoop() {
  clearTimeout(loopTimer);
  if (!currentStops.length) queuePlan();
  if (paused || document.hidden) return;
  playStop();
}

function playStop() {
  if (paused || document.hidden || !currentStops.length) return;
  const stop = currentStops[activeStopIndex];
  const room = $(`[data-room-id="${CSS.escape(stop.destinationId)}"]`);
  $$('.bank-room.active, .bank-vault.active, .bank-door.active').forEach((node) => node.classList.remove('active'));
  if (room) room.classList.add('active');
  const character = $('#pixel-character');
  const bubble = $('#thought-bubble');
  moveCharacterToDestination(room, character, bubble);
  character.classList.toggle('walking', stop.kind !== 'entrance' && stop.kind !== 'exit');
  bubble.textContent = stopMessage(stop);
  $('#loop-status').textContent = `${selectedCharacter()?.name || 'Character'}: ${stopMessage(stop)}`;
  $('#route-counter').textContent = `Stop ${activeStopIndex + 1} of ${currentStops.length}`;

  const delay = Math.max(550, 1700 / (state.animationSpeed || 1));
  loopTimer = setTimeout(() => {
    activeStopIndex += 1;
    if (activeStopIndex >= currentStops.length) {
      activeStopIndex = 0;
      if (pendingStops) { currentStops = pendingStops; pendingStops = null; renderBank(); }
    }
    playStop();
  }, delay);
}


function moveCharacterToDestination(room, character, bubble) {
  const scene = $('#bank-scene');
  if (!scene || !room || !character || !bubble) return;
  const sceneRect = scene.getBoundingClientRect();
  const roomRect = room.getBoundingClientRect();
  const x = Math.max(10, Math.min(sceneRect.width - 58, roomRect.left - sceneRect.left + roomRect.width / 2 - 24));
  const y = Math.max(30, Math.min(sceneRect.height - 86, roomRect.top - sceneRect.top + roomRect.height / 2 - 38));
  character.style.left = `${x}px`;
  character.style.top = `${y}px`;
  bubble.style.left = `${Math.max(8, Math.min(sceneRect.width - 290, x + 54))}px`;
  bubble.style.top = `${Math.max(8, y - 28)}px`;
}

function stopMessage(stop) {
  if (stop.kind === 'entrance') return 'Entering the bank with today’s plan.';
  if (stop.kind === 'exit') return 'Route complete. Returning to the entrance.';
  if (stop.kind === 'savings') return `Checking the Savings Vault: ${formatMoney(stop.amountCents)} saved.`;
  if (stop.amountCents > 0) return `Depositing ${formatMoney(stop.amountCents)} into ${stop.label}.`;
  return `${stop.label} still needs funding.`;
}


async function loadBankReferenceArt() {
  const image = $('#bank-reference-art');
  if (!image) return;
  try {
    const partCount = Number(image.dataset.bankArtParts || 4);
    const paths = Array.from({ length: partCount }, (_, index) => `./assets/approved-bank-reference-4k-${String(index + 1).padStart(2, '0')}.txt`);
    const parts = await Promise.all(paths.map(async (path) => {
      const response = await fetch(path);
      if (!response.ok) throw new Error(`Unable to load ${path}`);
      return response.text();
    }));
    image.src = `data:image/webp;base64,${parts.join('')}`;
  } catch {
    image.hidden = true;
    toast('The 4K bank artwork could not be loaded.');
  }
}
