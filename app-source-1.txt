function normalizeSavings(savings = {}) {
  const hasBoxes = Array.isArray(savings.boxes) && savings.boxes.length > 0;
  const boxes = DEFAULT_SAVINGS_BOXES.map((defaultBox, index) => {
    const savedBox = hasBoxes ? savings.boxes[index] : null;
    return {
      ...defaultBox,
      ...savedBox,
      id: savedBox?.id || defaultBox.id,
      name: savedBox?.name || defaultBox.name,
      targetCents: Math.max(0, Math.trunc(savedBox?.targetCents || 0)),
      savedCents: Math.max(0, Math.trunc(savedBox?.savedCents || 0)),
    };
  });
  if (!hasBoxes && (savings.targetCents > 0 || savings.savedCents > 0)) {
    boxes[0].targetCents = Math.max(0, Math.trunc(savings.targetCents || 0));
    boxes[0].savedCents = Math.max(0, Math.trunc(savings.savedCents || 0));
  }
  const summary = summarizeSavingsBoxes(boxes);
  return { name: savings.name || 'Savings Vault', boxes, ...summary };
}

function loadState() {
  try {
    const saved = JSON.parse(localStorage.getItem(STORAGE_KEY));
    if (!saved || !Array.isArray(saved.bills) || !Array.isArray(saved.characters)) return createInitialState();
    return { ...createInitialState(), ...saved, savings: normalizeSavings(saved.savings) };
  } catch {
    return createInitialState();
  }
}

function saveState(message = 'Saved locally') {
  state.savings = normalizeSavings(state.savings);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  $('#save-indicator').textContent = message;
  queuePlan();
}

function toast(message) {
  const node = $('#toast');
  node.textContent = message;
  node.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => node.classList.remove('show'), 2200);
}

}

function renderSummary() {
  const summary = summarizeBills(state.bills, state.asOfDate);
  const result = allocateAvailableFunds(state.bills, state.availableCents, state.asOfDate);
  const availableAfterBills = result.remainingCents;
  $('#summary-grid').innerHTML = [
    ['August bills', summary.totalCents, 'total'],
    ['Past due', summary.overdueCents, 'overdue'],
    ['Still outstanding', summary.outstandingCents, 'remaining'],
    ['Safe after plan', availableAfterBills, 'available'],
  ].map(([label, cents, kind]) => `<article class="summary-card ${kind}"><span>${label}</span><strong>${formatMoney(cents)}</strong></article>`).join('');
}

function renderFundingControls() {
  $('#available-funds').value = (state.availableCents / 100).toFixed(2);

  $('#active-character').innerHTML = state.characters.map((character) => `<option value="${escapeHtml(character.id)}" ${character.id === state.selectedCharacterId ? 'selected' : ''}>${escapeHtml(character.name)}</option>`).join('');
}

function renderBank() {
  const result = allocateAvailableFunds(state.bills, state.availableCents, state.asOfDate);
  currentAllocations = result.allocations;
  state.savings = normalizeSavings(state.savings);
  const selected = selectedCharacter();
  applyCharacterStyle($('#pixel-character'), selected);

  $('#bank-rooms').innerHTML = currentAllocations.map((allocation) => {
    const bill = state.bills.find((item) => item.id === allocation.billId);
    const status = allocation.status;
    const due = bill?.dueDate ? formatDate(bill.dueDate) : bill?.dueLabel || 'Recurring';
    return `<article class="bank-room bank-department ${status}" data-room-id="${escapeHtml(allocation.billId)}">
      <div class="room-header"><span class="room-name">${escapeHtml(allocation.name)}</span><span>${categoryIcon(bill?.category)}</span></div>
      <div class="room-screen">
        <div class="room-amount">${formatMoney(allocation.outstandingCents)}</div>
        <div>Due: ${escapeHtml(due)}</div>
        <div>Assigned: ${formatMoney(allocation.fundedCents)}</div>
        <span class="status-chip ${status}">${statusLabel(status)}</span>
      </div>
    </article>`;
  }).join('');
  renderVaultRoom();

  $('#funding-route').innerHTML = currentAllocations.map((allocation, index) => `<div class="route-item">
    <span class="route-index">${index + 1}</span>
    <div><strong>${escapeHtml(allocation.name)}</strong><div>${formatMoney(allocation.fundedCents)} assigned of ${formatMoney(allocation.outstandingCents)}</div></div>
    <span class="status-chip ${allocation.status}">${statusLabel(allocation.status)}</span>
  </div>`).join('');

  const attention = currentAllocations.filter((item) => ['overdue', 'partial', 'upcoming', 'due-today', 'recurring'].includes(item.status));
  $('#attention-list').innerHTML = attention.length ? attention.map((item) => `<div class="attention-item"><span>⚠</span><div><strong>${escapeHtml(item.name)}</strong><div>${item.status === 'overdue' ? 'Past due' : 'Needs funding'}: ${formatMoney(item.outstandingCents - item.fundedCents)}</div></div><span>${item.dueDate ? formatDate(item.dueDate) : 'Recurring'}</span></div>`).join('') : '<p>Every bill is fully covered.</p>';
}

  $('#vault-boxes').innerHTML = renderVaultBoxGrid(savings.boxes, true);

}

function renderBills() {
  $('#bill-count').textContent = `${state.bills.length} bills`;
  $('#bill-table-body').innerHTML = state.bills.map((bill) => {
    const status = classifyBill(bill, state.asOfDate);
    return `<tr>
      <td><strong>${escapeHtml(bill.name)}</strong><br><small>${escapeHtml(bill.category || 'Other')}</small></td>
      <td>${bill.dueDate ? formatDate(bill.dueDate) : escapeHtml(bill.dueLabel || 'Recurring')}</td>
      <td>${formatMoney(bill.amountCents)}</td><td>${formatMoney(bill.paidCents || 0)}</td>
      <td><span class="status-chip ${status}">${statusLabel(status)}</span></td>
      <td><div class="row-actions"><button class="small-button" data-edit-bill="${escapeHtml(bill.id)}" type="button">Edit</button><button class="small-button" data-delete-bill="${escapeHtml(bill.id)}" type="button">Delete</button></div></td>
    </tr>`;
  }).join('');
}

function renderCalendar() {
  const events = new Map();
  for (const bill of state.bills.filter((item) => item.dueDate?.startsWith('2026-08'))) {
    const day = Number(bill.dueDate.slice(-2));
    const list = events.get(day) || [];
    list.push(bill);
    events.set(day, list);
  }
  const cells = [];
  for (let blank = 0; blank < 6; blank++) cells.push('<div class="calendar-day empty"></div>');
  for (let day = 1; day <= 31; day++) {
    const dayEvents = events.get(day) || [];
    cells.push(`<div class="calendar-day"><div class="calendar-date">${day}</div>${dayEvents.map((bill) => `<div class="calendar-event ${classifyBill(bill, state.asOfDate)}">${escapeHtml(bill.name)}<br>${formatMoney(bill.amountCents)}</div>`).join('')}</div>`);
  }
  $('#calendar-grid').innerHTML = cells.join('');
  const sorted = [...state.bills].sort((a,b) => (a.dueDate || '9999').localeCompare(b.dueDate || '9999'));
  $('#agenda-list').innerHTML = `<div class="section-heading"><h3>August agenda</h3><span class="badge">${state.bills.length} items</span></div>` + sorted.map((bill) => `<div class="agenda-item"><span>${categoryIcon(bill.category)}</span><div><strong>${escapeHtml(bill.name)}</strong><div>${bill.dueDate ? formatDate(bill.dueDate) : escapeHtml(bill.dueLabel || 'Recurring')}</div></div><strong>${formatMoney(bill.amountCents)}</strong></div>`).join('');
}
